        // Lọc, tìm kiếm, phân trang và render bảng proxy
        function filterAndRenderTable() {
            let filtered = cachedProxies.slice();

            // 1. Lọc theo Quốc gia
            if (currentCountryFilter !== 'ALL') {
                filtered = filtered.filter(p => p.country === currentCountryFilter);
            }

            // 2. Lọc theo Trạng thái
            if (currentStatusFilter !== 'ALL') {
                filtered = filtered.filter(p => p.status === currentStatusFilter);
            }

            // 3. Tìm kiếm theo từ khóa
            if (currentSearchTerm) {
                filtered = filtered.filter(p =>
                    (p.host && p.host.toLowerCase().includes(currentSearchTerm)) ||
                    (p.port && p.port.toString().includes(currentSearchTerm)) ||
                    (p.countryName && p.countryName.toLowerCase().includes(currentSearchTerm)) ||
                    (p.city && p.city.toLowerCase().includes(currentSearchTerm)) ||
                    (p.isp && p.isp.toLowerCase().includes(currentSearchTerm)) ||
                    (p.username && p.username.toLowerCase().includes(currentSearchTerm))
                );
            }

            const totalFiltered = filtered.length;
            const totalPages = Math.ceil(totalFiltered / pageSize) || 1;
            if (currentPage > totalPages) currentPage = totalPages;
            if (currentPage < 1) currentPage = 1;

            const startIndex = (currentPage - 1) * pageSize;
            const endIndex = Math.min(startIndex + pageSize, totalFiltered);
            const pageItems = filtered.slice(startIndex, endIndex);

            // Cập nhật thông tin phân trang
            document.getElementById('paginationInfo').innerText = totalFiltered === 0
                ? 'Không tìm thấy proxy nào phù hợp'
                : `Hiển thị ${(startIndex + 1).toLocaleString()} - ${endIndex.toLocaleString()} của ${totalFiltered.toLocaleString()} proxy`;

            document.getElementById('paginationPageCurrent').innerText = `Trang ${currentPage} / ${totalPages}`;
            document.getElementById('btnPrevPage').disabled = currentPage <= 1;
            document.getElementById('btnNextPage').disabled = currentPage >= totalPages;

            const tbody = document.getElementById('realProxiesTableBody');
            if (pageItems.length === 0) {
                tbody.innerHTML = '<tr><td colspan="9" style="text-align: center; color: var(--text-muted); padding: 24px;">Không có dữ liệu proxy phù hợp với bộ lọc hiện tại.</td></tr>';
                updateMasterCheckbox();
                updateBatchActionsBar();
                return;
            }

            tbody.innerHTML = pageItems.map(p => {
                const flag = flagMap[p.country] || '🌐';
                const isSelected = selectedProxyIds.has(p.id);

                let statusBadge = `<span class="badge badge-unchecked">⚪ Chưa test</span>`;
                if (p.status === 'LIVE') {
                    statusBadge = `<span class="badge badge-live">🟢 LIVE</span>`;
                } else if (p.status === 'OFFLINE') {
                    statusBadge = `<span class="badge badge-offline">🔴 OFFLINE</span>`;
                }

                let pingHtml = '<span style="color: var(--text-muted);">-</span>';
                if (p.status === 'OFFLINE') {
                    pingHtml = `<span class="ping-tag ping-offline" style="background: rgba(239,68,68,0.2); color: #f87171; border: 1px solid rgba(239,68,68,0.4);">DIE</span>`;
                } else if (p.pingMs) {
                    let pingClass = p.pingMs < 100 ? 'ping-fast' : (p.pingMs < 250 ? 'ping-medium' : 'ping-slow');
                    pingHtml = `<span class="ping-tag ${pingClass}">${p.pingMs} ms</span>`;
                }

                const vpnActiveBadge = p.isActiveVpn
                    ? `<span class="badge badge-vpn-active">🎯 ĐANG CHẠY VPN</span>`
                    : `<button class="btn btn-outline btn-sm" onclick="selectVpnProxy('${p.id}')">Dùng cho VPN</button>`;

                const displayAddr = p.username ? `${p.host}:${p.port}:${p.username}:${p.password}` : `${p.host}:${p.port}`;

                return `
                    <tr class="${isSelected ? 'selected-row' : ''}">
                        <td style="text-align: center;">
                            <input type="checkbox" class="proxy-row-chk" data-id="${p.id}" ${isSelected ? 'checked' : ''} onchange="toggleSelectRow('${p.id}', this.checked)" style="cursor: pointer; width: 16px; height: 16px;">
                        </td>
                        <td>${statusBadge}</td>
                        <td><span class="badge badge-proto">${p.type.toUpperCase()}</span></td>
                        <td><strong>${flag} ${p.countryName}</strong> <span style="color: var(--text-muted); font-size: 12px;">(${p.city})</span></td>
                        <td>
                            <span class="code-pill">${displayAddr}</span>
                            <span class="copy-btn" title="Sao chép địa chỉ" onclick="navigator.clipboard.writeText('${displayAddr}')"> 📋</span>
                        </td>
                        <td>${pingHtml}</td>
                        <td style="color: var(--text-muted); font-size: 12px; max-width: 180px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;">${p.isp || 'Residential'}</td>
                        <td>${vpnActiveBadge}</td>
                        <td>
                            <div style="display: flex; gap: 4px;">
                                <button class="btn btn-sm btn-outline" onclick="testSingleProxy('${p.id}')">⚡ Test</button>
                                <button class="btn btn-sm btn-danger" onclick="deleteProxy('${p.id}')">🗑️</button>
                            </div>
                        </td>
                    </tr>
                `;
            }).join('');

            updateMasterCheckbox();
            updateBatchActionsBar();
        }

        // ==================== BATCH SELECTION & ACTIONS ====================
        function toggleSelectRow(id, isChecked) {
            if (isChecked) {
                selectedProxyIds.add(id);
            } else {
                selectedProxyIds.delete(id);
            }
            updateMasterCheckbox();
            updateBatchActionsBar();
        }

        function toggleSelectAll(masterEl) {
            const isChecked = masterEl.checked;
            const checkboxes = document.querySelectorAll('.proxy-row-chk');
            checkboxes.forEach(cb => {
                cb.checked = isChecked;
                const id = cb.getAttribute('data-id');
                if (id) {
                    if (isChecked) selectedProxyIds.add(id);
                    else selectedProxyIds.delete(id);
                }
            });
            updateBatchActionsBar();
        }

        function clearSelectedRows() {
            selectedProxyIds.clear();
            const chkAll = document.getElementById('chkSelectAll');
            if (chkAll) chkAll.checked = false;
            document.querySelectorAll('.proxy-row-chk').forEach(cb => cb.checked = false);
            updateBatchActionsBar();
        }

        function updateMasterCheckbox() {
            const master = document.getElementById('chkSelectAll');
            if (!master) return;
            const checkboxes = Array.from(document.querySelectorAll('.proxy-row-chk'));
            if (checkboxes.length === 0) {
                master.checked = false;
                master.indeterminate = false;
            } else {
                const checkedCount = checkboxes.filter(cb => cb.checked).length;
                master.checked = checkedCount === checkboxes.length;
                master.indeterminate = checkedCount > 0 && checkedCount < checkboxes.length;
            }
        }

        function updateBatchActionsBar() {
            const bar = document.getElementById('batchActionsBar');
            const countEl = document.getElementById('batchSelectedCount') || document.getElementById('selectedBatchCount');
            if (!bar) return;
            const count = selectedProxyIds.size;
            if (countEl) countEl.innerText = count.toLocaleString();
            if (count > 0) {
                bar.style.display = 'flex';
                bar.classList.add('active');
            } else {
                bar.style.display = 'none';
                bar.classList.remove('active');
            }
        }

        async function executeBatchDelete() {
            const count = selectedProxyIds.size;
            if (count === 0) return;
            if (!confirm(`Bạn có chắc chắn muốn xóa ${count} proxy đã chọn? Thao tác này không thể hoàn tác!`)) return;

            try {
                const ids = Array.from(selectedProxyIds);
                const res = await fetch('/api/v1/proxies/batch-delete', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ ids: ids })
                }).then(r => r.json());

                if (res.success) {
                    const deletedCount = res.data ? res.data.deleted : count;
                    cachedProxies = cachedProxies.filter(p => !selectedProxyIds.has(p.id));
                    selectedProxyIds.clear();
                    updateMasterCheckbox();
                    updateBatchActionsBar();
                    updateProxyStats(cachedProxies);
                    renderCountryPills(cachedProxies);
                    filterAndRenderTable();
                    alert(`✅ Đã xóa thành công ${deletedCount} proxy khỏi kho!`);
                } else {
                    alert('Lỗi: ' + (res.message || 'Không thể xóa proxy.'));
                }
            } catch (err) {
                alert('Lỗi kết nối khi xóa hàng loạt: ' + err.message);
            }
        }

        async function executeBatchTest() {
            const count = selectedProxyIds.size;
            if (count === 0) return;

            try {
                const ids = Array.from(selectedProxyIds);
                const res = await fetch('/api/v1/proxies/batch-test', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ ids: ids })
                }).then(r => r.json());

                if (res.success && res.data) {
                    const updatedMap = new Map(res.data.map(p => [p.id, p]));
                    for (let i = 0; i < cachedProxies.length; i++) {
                        if (updatedMap.has(cachedProxies[i].id)) {
                            cachedProxies[i] = updatedMap.get(cachedProxies[i].id);
                        }
                    }
                    updateProxyStats(cachedProxies);
                    filterAndRenderTable();
                    alert(`✅ Đã hoàn tất đo ping cho ${res.data.length} proxy!`);
                } else {
                    alert('Lỗi: ' + (res.message || 'Không thể test ping.'));
                }
            } catch (err) {
                alert('Lỗi kết nối khi test ping hàng loạt: ' + err.message);
            }
        }

        // ==================== AUTO HEALTH-CHECK (10 PHÚT) ====================
        function initHealthCheckPolling() {
            pollHealthCheckStatus();
            // Polling trạng thái mỗi 10 giây để đồng bộ
            setInterval(pollHealthCheckStatus, 10000);
            // Tick đếm ngược mỗi giây
            if (healthCheckIntervalTimer) clearInterval(healthCheckIntervalTimer);
            healthCheckIntervalTimer = setInterval(tickHealthCountdown, 1000);
        }

        async function pollHealthCheckStatus() {
            try {
                const res = await fetch('/api/v1/proxies/health-check/status').then(r => r.json());
                if (res && res.success && res.data) {
                    const data = res.data;
                    isHealthCheckRunning = data.isRunning;
                    healthCheckCountdownSec = Math.max(0, Math.floor(data.secondsUntilNextCheck));
                    updateHealthCheckWidget(data);
                }
            } catch (err) {
                console.warn("Không thể lấy trạng thái Health Check:", err);
            }
        }

        function tickHealthCountdown() {
            if (healthCheckCountdownSec > 0 && !isHealthCheckRunning) {
                healthCheckCountdownSec--;
            }
            const mins = Math.floor(healthCheckCountdownSec / 60);
            const secs = healthCheckCountdownSec % 60;
            const timeStr = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
            const timerEl = document.getElementById('healthCheckTimer');
            if (timerEl) {
                timerEl.innerText = isHealthCheckRunning ? 'Đang ping...' : timeStr;
            }
        }

        function updateHealthCheckWidget(data) {
            const statusEl = document.getElementById('healthCheckStatusText');
            const btnEl = document.getElementById('btnManualHealthCheck');
            const summaryEl = document.getElementById('healthCheckSummary');

            if (data.isRunning) {
                if (statusEl) statusEl.innerText = 'Đang quét tự động...';
                if (btnEl) {
                    btnEl.disabled = true;
                    btnEl.innerText = '⏳ Đang quét...';
                }
            } else {
                if (statusEl) statusEl.innerText = 'Tự động (Chu kỳ 10 phút)';
                if (btnEl) {
                    btnEl.disabled = false;
                    btnEl.innerText = '⚡ Ping Ngay';
                }
            }

            if (summaryEl) {
                const live = data.lastLiveCount || 0;
                const offline = data.lastOfflineCount || 0;
                const total = data.totalMonitored || (live + offline);
                summaryEl.innerText = `Sống: ${live.toLocaleString()} | Chết: ${offline.toLocaleString()} / Tổng: ${total.toLocaleString()}`;
            }
        }

        async function triggerManualHealthCheck() {
            const btn = document.getElementById('btnManualHealthCheck');
            if (btn) {
                btn.disabled = true;
                btn.innerText = '⏳ Đang khởi chạy...';
            }
            try {
                const res = await fetch('/api/v1/proxies/health-check/run', { method: 'POST' }).then(r => r.json());
                if (res && res.success) {
                    alert('Đã kích hoạt quét ping toàn bộ kho proxy! Hệ thống đang kiểm tra ngầm.');
                    await pollHealthCheckStatus();
                    setTimeout(fetchSystemData, 3000);
                } else {
                    alert('Thông báo: ' + (res.message || 'Không thể kích hoạt.'));
                }
            } catch (err) {
                alert('Lỗi kích hoạt health check: ' + err.message);
            } finally {
                if (btn) {
                    btn.disabled = false;
                    btn.innerText = '⚡ Ping Ngay';
                }
            }
        }

        function generateProxies() {
            const tenant = document.getElementById('genTenantSelect').value;
            const country = document.getElementById('genCountry').value;
            const mode = document.getElementById('genMode').value;
            const proto = document.getElementById('genProtocol').value;
            const qty = parseInt(document.getElementById('genQuantity').value) || 10;

            const results = [];
            const gwHost = "127.0.0.1";
            const gwPort = 10000;

            for (let i = 0; i < qty; i++) {
                let userStr = tenant;
                if (country !== 'ALL') userStr += `-country-${country.toLowerCase()}`;
                if (mode === 'sticky') {
                    const sess = Math.random().toString(36).substring(2, 8);
                    userStr += `-session-${sess}-time-30`;
                }
                const pass = "nextaipass";
                results.push(`${proto}://${userStr}:${pass}@${gwHost}:${gwPort}`);
            }

            document.getElementById('genResult').value = results.join('\n');
        }

        function renderTenantsTable(tenants) {
            const tbody = document.getElementById('tenantsTableBody');
            tbody.innerHTML = tenants.map(t => {
                const usedMb = (t.usedBytes / (1024 * 1024)).toFixed(1);

        let cachedXrayServers = [];
        let cachedXrayUsers = [];

        async function fetchXrayVpnData() {
            try {
                const [serversRes, usersRes] = await Promise.all([
                    fetch('/api/xray/servers').then(r => r.json()).catch(() => null),
                    fetch('/api/xray/users').then(r => r.json()).catch(() => null)
                ]);

                if (serversRes) {
                    cachedXrayServers = Array.isArray(serversRes) ? serversRes : (serversRes.value || []);
                    renderXrayServersTable(cachedXrayServers);
                }

                if (usersRes) {
                    cachedXrayUsers = Array.isArray(usersRes) ? usersRes : (usersRes.value || []);
                    renderXrayUsersTable(cachedXrayUsers);
                }
            } catch (err) {
                console.error("Lỗi nạp dữ liệu Xray VPN:", err);
            }
        }

        function renderXrayServersTable(servers) {
            const tbody = document.getElementById('xrayServersTableBody');
            if (!tbody) return;

            // Chỉ nạp và hiển thị những server có thể dùng làm VPN
            const usableServers = servers.filter(s => s.vpn_usable === 1 || s.vpn_usable === true || s.status === 'ONLINE');
            
            const total = servers.length;
            const usable = usableServers.length;
            const usableEl = document.getElementById('valXrayUsableCount');
            if (usableEl) usableEl.innerText = `${usable} / ${total} Online (${total > 0 ? Math.round((usable/total)*100) : 0}%)`;

            const pings = usableServers.map(s => s.ping_ms).filter(p => p > 0);
            const avg = pings.length > 0 ? Math.round(pings.reduce((a,b)=>a+b,0)/pings.length) : 0;
            const pingEl = document.getElementById('valXrayAvgPing');
            if (pingEl) pingEl.innerText = avg > 0 ? `~${avg} ms` : 'N/A';

            if (usableServers.length === 0) {
                tbody.innerHTML = `<tr><td colspan="8" style="text-align: center; padding: 24px; color: var(--danger);">Không có máy chủ nào đủ điều kiện hoạt động làm VPN.</td></tr>`;
                return;
            }

            const flagIcons = { 'bg': '🇧🇬', 'de': '🇩🇪', 'vn': '🇻🇳', 'us': '🇺🇸', 'sg': '🇸🇬' };

            tbody.innerHTML = usableServers.map(s => {
                const flag = flagIcons[s.country_code.toLowerCase()] || '🌐';
                const connectAddr = s.connect_address || s.host || s.ip;
                const isDnsBypass = s.dns_status === 'DNS_FAILED_USE_DIRECT_IP' || s.id === 3;
                
                return `
                    <tr>
                        <td><strong>#${s.id}</strong></td>
                        <td>
                            <span style="font-size: 18px; vertical-align: middle; margin-right: 6px;">${flag}</span>
                            <strong>${s.country_name || s.country_code.toUpperCase()}</strong> - ${s.city || 'Default'}
                        </td>
                        <td>
                            <span class="code-pill" style="font-weight: 700; color: #38bdf8;">${connectAddr}</span>
                            ${isDnsBypass ? '<br><span style="font-size: 10px; color: #f59e0b;">⚠️ IP Trực Tiếp (Bypass DNS lỗi)</span>' : ''}
                        </td>
                        <td><span class="code-pill">${s.vless_port}</span></td>
                        <td>
                            <span class="badge badge-proto">VLESS Reality</span>
                            <div style="font-size: 11px; color: var(--text-muted); margin-top: 2px;">SNI: <span style="color: #67e8f9;">${s.xsni}</span></div>
                        </td>
                        <td>
                            <span class="ping-tag ${s.ping_ms < 100 ? 'ping-fast' : s.ping_ms < 300 ? 'ping-medium' : 'ping-slow'}">
                                ⚡ ${s.ping_ms} ms
                            </span>
                        </td>
                        <td>
                            <span class="badge badge-live">🟢 SẴN SÀNG LÀM VPN</span>
                        </td>
                        <td style="text-align: right; white-space: nowrap;">
                            <button class="btn btn-sm" onclick="copyVlessUrl('${s.vless_url}')" style="background: linear-gradient(135deg, #2563eb, #06b6d4);">
                                📋 Copy Link
                            </button>
                            <button class="btn btn-outline btn-sm" onclick="testSingleServer(${s.id})">
                                ⚡ Test
                            </button>
                        </td>
                    </tr>
                `;
            }).join('');
        }

        function copyVlessUrl(url) {
            if (!url) {
                alert('Không có link VLESS khả dụng.');
                return;
            }
            navigator.clipboard.writeText(url).then(() => {
                alert('✅ Đã sao chép link VLESS Reality vào bộ nhớ tạm!\nBạn có thể dán vào KikiLogin, V2RayN, Shadowrocket, Clash Meta để kết nối.');
            }).catch(() => {
                prompt('Copy link VLESS tại đây:', url);
            });
        }

        function copyAllVlessUrls() {
            if (!cachedXrayServers || cachedXrayServers.length === 0) return;
            const usable = cachedXrayServers.filter(s => s.vpn_usable === 1 || s.vpn_usable === true);
            const urls = usable.map(s => s.vless_url).filter(Boolean);
            if (urls.length === 0) {
                alert('Không có link VLESS nào khả dụng.');
                return;
            }
            const text = urls.join('\n');
            navigator.clipboard.writeText(text).then(() => {
                alert(`✅ Đã sao chép toàn bộ ${urls.length} link VLESS Reality vào bộ nhớ tạm!`);
            }).catch(() => {
                prompt('Copy các link VLESS tại đây:', text);
            });
        }

        function copyTextToClipboard(text, msg) {
            navigator.clipboard.writeText(text).then(() => {
                alert('✅ ' + (msg || 'Đã sao chép vào bộ nhớ tạm!'));
            });
        }
